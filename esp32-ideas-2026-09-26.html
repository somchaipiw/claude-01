นวัตกรรม ESP32 ล่าสุด

ESP32-S31 รัน Linux 6.18 / 7.1 ได้แล้ว! BREAKING

เหตุการณ์สำคัญครั้งประวัติศาสตร์ — ESP32-S31 กลายเป็น ESP32 ตัวแรกที่รัน Linux kernel เต็มรูปแบบได้ เนื่องจากมี MMU (Memory Management Unit) แท้จริง ซึ่งรุ่นก่อนหน้าทั้งหมดไม่มี Espressif ออก Linux BSP Developer Preview บน Buildroot 2025.02 พร้อม U-Boot

Linux 6.18 Linux 7.1 MMU RISC-V Developer Preview
ทำไมถึงสำคัญมาก? Linux kernel ต้องการ MMU สำหรับ virtual memory และ process isolation ESP32-P4 ยังใช้แค่ PMP (Physical Memory Protection) ทำให้รัน Linux จริงไม่ได้ แต่ ESP32-S31 มี MMU เต็มรูปแบบ — เปลี่ยนเกมทั้งหมด
สเปค ESP32-S31 ที่ทำให้ Linux เป็นไปได้
  • CPU Dual-core RISC-V @ 320 MHz
  • MMU ✅ มี (ต่างจาก ESP32-P4 ที่ใช้ PMP)
  • Wi-Fi Wi-Fi 6 (802.11ax)
  • Ethernet Gigabit Ethernet ในตัว
  • Linux ver. 6.18 (ไดร์เวอร์ครบ) / 7.1 (LCD 800×480)
  • สถานะ Developer Preview (ไม่สำหรับ Production)
  • Buildroot 2025.02 + U-Boot
⚡ ใครทำได้ก่อน? นักพัฒนา port Linux 7.1 ให้แสดง console บน LCD 800×480 ได้แล้ว — ใกล้ Raspberry Pi Zero มากขึ้นทุกที

Linux 6.11 บน ESP32-S3 "ดั้งเดิม" — ทำได้ด้วย Trick

นักพัฒนา port Linux 6.11 ให้รันบน ESP32-S3 รุ่นเก่าได้ แม้ไม่มี MMU ต้องใช้เทคนิคพิเศษ — รันใน noMMU mode โดย kernel ทำงานใน flat address space ไม่รองรับ multi-process แต่ใช้เป็น demo/proof of concept ได้

ESP32-S3 Linux 6.11 noMMU mode PoC Only
ข้อจำกัด: ไม่มี virtual memory, ไม่รองรับ process isolation, รัน application เดียวกันเวลาเดียว — แต่เปิด shell ได้! Hackaday ลงข่าวเมื่อ 15 ก.ย. 2026

ESP32-S31 Linux MQTT Node-RED

  • ติดตั้ง Mosquitto broker บน Linux filesystem

VEML6075 — UV Index Sensor ราคาถูกสำหรับ ESP32

Vishay VEML6075 วัด UVA และ UVB แยกกัน คำนวณ UV index อัตโนมัติ ใช้ I2C ทำงานได้กับ 3.3V โดยตรง เหมาะสำหรับ outdoor weather station และ smart terrarium ที่ต้องควบคุมแสง UV สำหรับสัตว์เลื้อยคลาน

I2C 3.3V UVA + UVB
  • วัด UVA 315-400nm, UVB 280-315nm
  • Output UV Index 0-11+
  • Interface I2C (addr 0x10)
  • ราคา ~$2-4 (breakout board)

Seeed mmWave Library v2.0-RC — รองรับ 60GHz ครบทุกรุ่น

Seeed Studio ปล่อย v2.0.0-rc สำหรับ Arduino/ESP32 รองรับ MR60BHA2 และ MR60FDA2 บน XIAO ESP32-C6 มี API ง่ายขึ้นมาก parse ข้อมูล fall/breath/heart ได้ทันที ไม่ต้องเขียน protocol เอง

Arduino XIAO ESP32-C6 v2.0-rc GitHub 
// ติดตั้ง: GitHub Love4yzp/Seeed-mmWave-library
#include "Seeed_mmWave.h"
MR60BHA2 mmWave(&Serial1);

void loop() {
  mmWave.update();
  float bpm = mmWave.heartRate;
  float rr  = mmWave.breathRate;
  Serial.printf("HR: %.1f BPM, RR: %.1f\n", bpm, rr);
}

Smart Terrarium — ควบคุม Relay ตามอุณหภูมิ/ความชื้น

#include <DHT.h>
DHT dht(4, DHT22);

const int RELAY_HEAT = 12;
const int RELAY_FOG  = 13;
float TARGET_TEMP = 28.0;  // °C
float TARGET_HUM  = 65.0;  // %

void loop() {
  float t = dht.readTemperature();
  float h = dht.readHumidity();

  // ควบคุมหลอดความร้อน
  digitalWrite(RELAY_HEAT, t < TARGET_TEMP);
  // ควบคุมหมอกพ่น
  digitalWrite(RELAY_FOG,  h < TARGET_HUM);

  delay(5000);
}

Waste Monitor — วัดระดับขยะ + ส่ง MQTT

// วัดระดับขยะด้วย Ultrasonic
long getWasteLevel() {
  digitalWrite(TRIG, LOW); delayMicroseconds(2);
  digitalWrite(TRIG, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG, LOW);
  long dur = pulseIn(ECHO, HIGH);
  long dist = dur * 0.034 / 2; // cm
  int fill = map(dist, BIN_DEPTH, 5, 0, 100);
  return constrain(fill, 0, 100);
}

void loop() {
  int pct = getWasteLevel();
  // ส่ง MQTT เมื่อขยะเกิน 80%
  if (pct > 80) {
    mqtt.publish("bin/alert",
      String(pct).c_str());
  }
  delay(60000); // ทุก 1 นาที
}

OpenMote-style — IR + BLE ด้วย ESP32-S3

#include <IRremoteESP8266.h>
#include <BleKeyboard.h>

IRsend irsend(4);   // IR LED pin
BleKeyboard bleKb("OpenMote");

void setup() {
  irsend.begin();
  bleKb.begin();
}

void loop() {
  if (digitalRead(BTN_A) == LOW) {
    // ส่ง IR ปิด/เปิด TV Samsung
    irsend.sendSAMSUNG(0xE0E040BF, 32);
  }
  if (digitalRead(BTN_B) == LOW) {
    // ส่ง BLE media play/pause
    bleKb.write(KEY_MEDIA_PLAY_PAUSE);
  }
}
