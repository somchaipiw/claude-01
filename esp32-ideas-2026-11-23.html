ESP32-P4 Chip Rev v3
ESP32-P4 Chip Revision v3.x — อัปเกรดประสิทธิภาพสำคัญ (พ.ค. 2026)
Espressif อัปเกรด ESP32-P4 revision ใหม่ เพิ่มความสามารถระดับฮาร์ดแวร์หลายด้านสำหรับงาน HMI และ multimedia
  • Clock สูงสุดจาก 360 MHz → 400 MHz
  • เพิ่ม Zb Bit-Manipulation Extension สำหรับประมวลผลข้อมูลเร็วขึ้น
  • Dedicated Hardware RNG ที่ปลอดภัยกว่าเดิม
  • 160 MHz I2S Clock Source รองรับ AES67 Audio-over-IP
  • เหมาะกับ Matter 1.5 Smart Home Camera
Use case หลัก: Touchscreen HMI, Large MIPI DSI Display, NuttX Web Panel

Water Quality I2C/Analog
Atlas Scientific EZO-pH + EZO-EC
โปรบ pH และ EC (Electrical Conductivity/TDS) ระดับ industrial ที่เชื่อมต่อผ่าน I2C โดยตรง ไม่ต้องวงจร analog เพิ่ม เหมาะกับ hydroponics และ aquaponics
  • EZO-pH: ช่วง 0–14 pH, ความแม่นยำ ±0.001 pH
  • EZO-EC: วัด EC, TDS, Salinity, SG
  • Calibration ง่ายผ่าน I2C command
  • งบสูง (~$50–100/ชิ้น) แต่ทนทานกว่า clone มาก
ใช้คู่กับ DS18B20 temperature สำหรับ temperature compensation ของ EC

Biometric I2C
MAX30102 — Heart Rate & SpO2 Sensor
IC วัด heart rate และออกซิเจนในเลือด (SpO2) ผ่าน Photoplethysmography (PPG) เชื่อมต่อผ่าน I2C ใช้ในงาน wearable ได้ดี
  • ใช้ไฟ 1.8–3.3V, กระแสต่ำมาก (600 µA ขณะวัด)
  • SpO2 accuracy ±2% (เทียบกับ Contec CMS50D)
  • Library: SparkFun_MAX3010x_Sensor_Library
  • espressif/esp_health component รองรับ MAX30102 แล้ว
⚠️ ค่าที่อ่านได้เป็น experimental grade เท่านั้น ไม่ผ่านมาตรฐาน medical ใช้เพื่อ hobby/research เท่านั้น

ECG SPI/I2C
MAX30003 — Single-Channel ECG AFE
IC สำหรับ Electrocardiogram (ECG) front-end วัดสัญญาณหัวใจ 1 ช่อง เชื่อมต่อผ่าน SPI ให้ค่าความละเอียดสูง
  • 18-bit ADC, Noise < 9 µV rms
  • Built-in R-to-R detector, Lead-off detection
  • ใช้ไฟ 1.8V core + 1.8–3.3V digital I/O
  • ใช้คู่ MAX30102 (PPG) + LSM6DS3 (IMU) ในบอร์ด wearable

Flow Sensor Digital
YF-S201 / FS300A — Water Flow Sensor สำหรับ Hydroponics
flow sensor แบบ Hall-effect ราคาประหยัดสำหรับวัดปริมาณน้ำที่ไหลผ่านท่อ เหมาะกับระบบ hydroponics และ aquaponics
  • Output: Pulse signal (Hall effect) อ่านด้วย GPIO interrupt
  • YF-S201: ช่วง 1–30 L/min, Pulse: ~450 pulses/liter
  • FS300A: ช่วง 10–200 L/min สำหรับท่อใหญ่
  • ติดตั้งง่าย เชื่อมต่อ ESP32 GPIO + 10kΩ pull-up
ใช้ ESP32 MCPWM หรือ PCNT hardware counter เพื่อนับ pulse แม่นยำกว่า software interrupt
🌡️
DO Sensor Analog
Dissolved Oxygen (DO) Probe — Atlas Scientific EZO-DO
sensor วัดออกซิเจนละลายน้ำ สำคัญมากสำหรับระบบ aquaponics เพราะปลาต้องการ DO ≥ 5 mg/L
  • ช่วงวัด: 0–35.99 mg/L
  • Accuracy: ±0.05 mg/L
  • Interface: I2C (EZO circuit) หรือ UART
  • ต้อง calibrate กับน้ำอิ่มตัวออกซิเจน (100% air saturation)
ค่า DO ขึ้นอยู่กับอุณหภูมิน้ำ — ใช้ DS18B20 คู่กันเสมอ

esp32s3_face_detect.ino — ESP-DL + OV2640 Camera (Arduino)
C++ / Arduino 
#include "esp_camera.h"
#include "dl_detect_mnp.hpp"   // ESP-DL MobileNet face detector
#include "dl_image.hpp"

// LILYGO T-Camera S3 camera pin config
#define PWDN_GPIO_NUM  -1
#define RESET_GPIO_NUM -1
#define XCLK_GPIO_NUM  8
#define SIOD_GPIO_NUM  2
#define SIOC_GPIO_NUM  3
#define Y9_GPIO_NUM    37
#define Y8_GPIO_NUM    36
#define Y7_GPIO_NUM    35
#define Y6_GPIO_NUM    34
#define Y5_GPIO_NUM    33
#define Y4_GPIO_NUM    17
#define Y3_GPIO_NUM    15
#define Y2_GPIO_NUM    14
#define VSYNC_GPIO_NUM 6
#define HREF_GPIO_NUM  7
#define PCLK_GPIO_NUM  13

dl::detect::MNP detector;   // ESP-DL face detection model

void setup() {
  Serial.begin(115200);

  camera_config_t config;
  config.ledc_channel = LEDC_CHANNEL_0;
  config.ledc_timer   = LEDC_TIMER_0;
  config.pin_d0       = Y2_GPIO_NUM;
  // ... (ใส่ pin ที่เหลือตามตัวแปรด้านบน) ...
  config.xclk_freq_hz = 20000000;
  config.pixel_format = PIXFORMAT_RGB565;
  config.frame_size   = FRAMESIZE_240X240;   // 240x240 สำหรับ inference
  config.fb_count     = 2;
  config.fb_location  = CAMERA_FB_IN_PSRAM;   // ใช้ PSRAM
  config.grab_mode    = CAMERA_GRAB_LATEST;

  esp_err_t err = esp_camera_init(&config);
  if (err != ESP_OK) {
    Serial.printf("Camera init failed: 0x%x\n", err);
    return;
  }
  Serial.println("Camera ready — ESP-DL face detect running");
}

void loop() {
  camera_fb_t *fb = esp_camera_fb_get();
  if (!fb) return;

  // สร้าง Tensor จาก frame buffer
  dl::image::img_t img;
  img.data   = (uint16_t *)fb->buf;
  img.width  = fb->width;
  img.height = fb->height;
  img.pix_type = dl::image::DL_IMAGE_PIX_TYPE_RGB565;

  // รัน inference
  auto results = detector.infer(&img);

  Serial.printf("พบใบหน้า: %d ใบ\n", results.size());
  for (auto &r : results) {
    Serial.printf("  box=[%d,%d,%d,%d] score=%.2f\n",
      r.box[0], r.box[1], r.box[2], r.box[3], r.score);
  }

  esp_camera_fb_return(fb);
  delay(33);  // ~30 fps target (จริงๆ ได้ 10–15 fps)
}
hydroponics_ph_ec.ino — ESP32 + Atlas Scientific EZO (I2C)
C++ / Arduino 
#include "Wire.h"

#define PH_ADDR   0x63   // Atlas EZO-pH default I2C address
#define EC_ADDR   0x64   // Atlas EZO-EC default I2C address

float readEZO(uint8_t addr, const char* cmd, int waitMs) {
  Wire.beginTransmission(addr);
  Wire.print(cmd);
  Wire.endTransmission();
  delay(waitMs);

  Wire.requestFrom(addr, (uint8_t)20);
  Wire.read();  // response code (1 = success)
  char buf[20] = {0};
  int i = 0;
  while (Wire.available()) buf[i++] = Wire.read();
  return atof(buf);
}

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22);  // SDA=21, SCL=22

  // ตั้ง continuous mode ปิด ใช้ single-read แทน
  Wire.beginTransmission(PH_ADDR); Wire.print("C,0"); Wire.endTransmission();
  Wire.beginTransmission(EC_ADDR); Wire.print("C,0"); Wire.endTransmission();
  delay(300);
}

void loop() {
  float ph = readEZO(PH_ADDR, "R", 900);    // รอ 900ms สำหรับ pH
  float ec = readEZO(EC_ADDR, "R", 600);    // รอ 600ms สำหรับ EC

  Serial.printf("pH: %.3f | EC: %.2f µS/cm\n", ph, ec);

  // ตรรกะ dosing pump ตัวอย่าง
  if (ph > 6.5) {
    Serial.println("pH สูงเกิน — เปิดปั๊ม pH-down");
    // digitalWrite(PUMP_PH_DOWN_PIN, HIGH);
  } else if (ph < 5.5) {
    Serial.println("pH ต่ำเกิน — เปิดปั๊ม pH-up");
  }

  delay(10000);  // วัดทุก 10 วินาที
}
