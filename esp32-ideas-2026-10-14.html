ESP32-C61 — RISC-V + Wi-Fi 6 + BLE 5 ราคาประหยัด
Mass Production 2025 | จัดจำหน่าย 2026
Espressif เปิดตัว ESP32-C61 ให้เป็นซีรีส์ SoC รุ่นประหยัดสำหรับ IoT ที่ต้องการ Wi-Fi 6 และ Bluetooth 5 LE พร้อมกัน บน RISC-V Core 32-bit ทำงานที่ 160 MHz รองรับฟีเจอร์ Wi-Fi 6 ครบ เช่น OFDMA, MU-MIMO, Target Wake Time (TWT) สำหรับการประหยัดพลังงานสูงสุด

จุดเด่นสำคัญ: รองรับ External PSRAM, Bluetooth Mesh 1.1, Extended Advertising และ Coded PHY — เหมาะกับ Matter, Zigbee-friendly mesh และ BLE IoT ระดับ production

🔋
Target Wake Time (TWT) — ประหยัดพลังงานใน Wi-Fi 6
ฟีเจอร์ที่เปลี่ยนเกม IoT ใช้ถ่านได้นานขึ้น
TWT (Target Wake Time) คือฟีเจอร์สำคัญใน Wi-Fi 6 ที่ให้ AP กำหนดเวลาที่ชัดเจนว่า device ควร "ตื่น" เพื่อส่งรับข้อมูล แล้วกลับไป deep sleep ในเวลาที่เหลือ ผลลัพธ์คือ ESP32-C61 ใช้พลังงานน้อยกว่า Wi-Fi 4 มาก เหมาะกับ:
  • 1เซ็นเซอร์ IoT ที่ส่งข้อมูลทุก 5–60 นาที
  • 2Smart meter ที่ต้องการอายุแบตเตอรี่ 3–5 ปี
  • 3Asset tracker ในโรงงานหรือโกดัง
  • 4อุปกรณ์ Matter ที่ประหยัดพลังงาน
ทำไมถึงสำคัญกว่า ESP32-C6

ESP32-C61 ถูกออกแบบมาให้ BOM ราคาต่ำกว่า C6 เหมาะสำหรับ mass production ในปริมาณมาก ขณะที่ยังคงสเปก Wi-Fi 6 ครบถ้วน

💻 ตัวอย่าง Code — อ่าน TDS + Turbidity ด้วย ADS1115

ตัวอย่างนี้ใช้ ADS1115 อ่านค่าเซ็นเซอร์ TDS และ Turbidity พร้อมกัน พร้อม temperature compensation จาก DS18B20 (ที่ channel 3)
#include <Wire.h> #include <Adafruit_ADS1X15.h> #include <OneWire.h> #include <DallasTemperature.h> Adafruit_ADS1115 ads; OneWire oneWire(4); DallasTemperature tempSensor(&oneWire); void setup() { Serial.begin(115200); ads.begin(); // I2C address 0x48 (ADDR pin to GND) ads.setGain(GAIN_ONE); // ±4.096V range tempSensor.begin(); } float readTDS(float voltage, float temperature) { float compensationCoeff = 1.0 + 0.02 * (temperature - 25.0); float compVoltage = voltage / compensationCoeff; return (133.42 * pow(compVoltage, 3) - 255.86 * pow(compVoltage, 2) + 857.39 * compVoltage) * 0.5; } float readTurbidity(float voltage) { if (voltage < 2.5) return 3000.0; // Max NTU return -1120.4 * sq(voltage) + 5742.3 * voltage - 4353.8; } void loop() { tempSensor.requestTemperatures(); float temp = tempSensor.getTempCByIndex(0); float tdsV = ads.readADC_SingleEnded(0) * 0.1875 / 1000.0; float turbidV = ads.readADC_SingleEnded(1) * 0.1875 / 1000.0; float tds = readTDS(tdsV, temp); float turbidity = readTurbidity(turbidV); Serial.printf("Temp: %.1f°C | TDS: %.0f ppm | Turbidity: %.0f NTU\n", temp, tds, turbidity); delay(5000); }
