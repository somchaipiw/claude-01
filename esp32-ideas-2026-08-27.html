Spotlight Sensors
📡
DW1000 / DWM3000
UWB Ranging & Positioning chip จาก Qorvo — ส่ง pulse ความกว้าง 500 MHz ผ่าน 3.5–6.5 GHz คำนวณ Time-of-Flight ด้วยความละเอียด 15 ps (picosecond) ทำให้วัดระยะได้ ±10 cm
±10 cm SPI Interface 3.3V ~500 MHz BW
⚡
ADXL345
3-Axis MEMS Accelerometer ราคาถูก สำหรับ vibration analysis — range ±16g, output rate สูงสุด 3,200 Hz, รองรับ SPI และ I2C ใช้ร่วมกับ ESP32 FFT เพื่อตรวจจับ bearing fault
3200 Hz ODR ±16g I2C / SPI 3.3V / 5V
🔑
PN532 NFC Module
NFC/RFID reader/writer รองรับ ISO 14443A/B, Mifare, FeliCa, และ ISO 18092 (P2P) สื่อสารผ่าน I2C, SPI, หรือ UART — ระยะอ่านสูงสุด 5-7 cm เหมาะสำหรับ Smart Lock และ NFC tag programming
NFC P2P Mifare I2C/SPI/UART 5-7cm Range
🌫️
BME688
4-in-1 Environmental sensor จาก Bosch: Temperature, Humidity, Pressure, และ Gas/VOC พร้อม AI feature extraction onboard ด้วย BSEC 2.x library — ตรวจ IAQ (Indoor Air Quality Index) ได้โดยตรง ไม่ต้องคำนวณเอง

Arduino / C++
predictive_maintenance.ino — FFT Vibration Analysis บน ESP32 
// ESP32 Predictive Maintenance — FFT Vibration Analysis
// Hardware: ESP32 + ADXL345 (SPI) + MQTT Alert
#include <SPI.h>
#include <WiFi.h>
#include <PubSubClient.h>
#include <arduinoFFT.h>

#define SAMPLES       1024
#define SAMPLING_HZ  3200
#define ADXL_CS      5

double vReal[SAMPLES], vImag[SAMPLES];
float  baselineRMS = 0;        // เก็บ baseline ตอน normal operation

ArduinoFFT FFT = ArduinoFFT(vReal, vImag, SAMPLES, SAMPLING_HZ);
WiFiClient     espClient;
PubSubClient   mqtt(espClient);

float readADXL345_Z() {
  // อ่าน Z-axis acceleration จาก ADXL345 ผ่าน SPI
  digitalWrite(ADXL_CS, LOW);
  SPI.transfer(0x36 | 0xC0);  // register DATAZ0 + read + multi
  int16_t raw = SPI.transfer16(0);
  digitalWrite(ADXL_CS, HIGH);
  return raw * 0.0039f;          // แปลงเป็น g (4mg/LSB @ ±2g)
}

void analyzeVibration() {
  unsigned long ts = micros();
  for (int i = 0; i < SAMPLES; i++) {
    while (micros() - ts < (unsigned long)i * (1000000/SAMPLING_HZ));
    vReal[i] = readADXL345_Z();
    vImag[i] = 0;
  }

  FFT.windowing(FFT_WIN_TYP_HANN, FFT_FORWARD);
  FFT.compute(FFT_FORWARD);
  FFT.complexToMagnitude();

  // คำนวณ RMS และหา dominant frequency
  double rms = 0, peakFreq = FFT.majorPeak();
  for (int i = 1; i < SAMPLES/2; i++) rms += vReal[i] * vReal[i];
  rms = sqrt(rms / (SAMPLES/2));

  // ตรวจสอบความผิดปกติ — ถ้า RMS เกิน 3x baseline แจ้งเตือน
  if (baselineRMS > 0 && rms > baselineRMS * 3.0f) {
    char msg[128];
    sprintf(msg,
      "{\"rms\":%.3f,\"peak_hz\":%.1f,\"baseline\":%.3f,\"ratio\":%.1f}",
      rms, peakFreq, baselineRMS, rms/baselineRMS);
    mqtt.publish("factory/pump01/alert", msg);
  }
}
uwb_anchor.ino — ESP32 + DW1000 Anchor Node 
// ESP32 UWB Anchor Node — รับ ranging request จาก Tag
// Library: DW1000 by thotro (Arduino)
#include <SPI.h>
#include <DW1000.h>
#include <DW1000Ranging.h>
#include <WiFi.h>
#include <ArduinoJson.h>

#define ANCHOR_ID  "01"          // ID ของ anchor node นี้
#define ANCHOR_X   0.0f         // ตำแหน่ง X จริงในอาคาร (เมตร)
#define ANCHOR_Y   0.0f         // ตำแหน่ง Y จริงในอาคาร (เมตร)

void newRangeHandler() {
  DW1000Device* dev = DW1000Ranging.getDistantDevice();
  float dist = dev->getRange();   // ระยะทาง (เมตร)

  // ส่งระยะทางไปยัง RTLS server ผ่าน UDP
  StaticJsonDocument<128> doc;
  doc["anchor"]  = ANCHOR_ID;
  doc["tag"]     = dev->getShortAddress();
  doc["dist_m"]  = (int)(dist * 1000) / 1000.0f;
  doc["ax"]      = ANCHOR_X;
  doc["ay"]      = ANCHOR_Y;
  doc["ts"]      = millis();

  char buf[128];
  serializeJson(doc, buf);
  // udp.broadcastTo(buf, SERVER_IP, 5000);
}

void setup() {
  DW1000Ranging.initCommunication(SPI_CS, DW1000.SELECT_PIN(27), 34);
  DW1000Ranging.attachNewRange(newRangeHandler);
  DW1000Ranging.startAsAnchor("82:17:5B:D5:A9:9A:E2:9C",
                               DW1000.TRX_RATE_6800KBPS);
}

void loop() {
  DW1000Ranging.loop();
}

Power Optimization
📐 คำนวณอายุแบตเตอรี่ E-Ink Node
1
Active (อัปเดต display): ~80 mA × 5 วินาที = 0.111 mAh
2
Deep Sleep (ส่วนใหญ่): 0.022 mA × 3,595 วินาที = 21.97 mAh/วัน
3
อัปเดตทุกชั่วโมง: 0.111 × 24 + 21.97 ≈ 24.6 mAh/วัน
4
แบต 18650 (3,000 mAh): 3000 / 24.6 ≈ 122 วัน (~4 เดือน)
