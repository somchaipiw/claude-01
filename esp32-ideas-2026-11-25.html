ESP32-H21 — Chip ใหม่สำหรับ Wireless Mesh IoT
Espressif ประกาศ ESP32-H21 เมื่อ 3 มีนาคม 2026 เป็นชิปสำหรับ Thread, Zigbee, Matter, และ Bluetooth LE โดยเฉพาะ ไฮไลต์คือมี on-chip DC-DC converter ทำให้กินไฟน้อยลงอย่างมาก เหมาะกับ sensor node ที่ใช้พลังงานจากแบตเตอรี่

สเปคเด่น: RISC-V 32-bit core, IEEE 802.15.4 radio + BLE 5.2, กระแสรับสัญญาณ ~8.2 mA, Deep sleep ต่ำสุด ~5 µA, ส่งสัญญาณได้ถึง 20 dBm รองรับ ESP-IDF และ ESP-Matter อย่างเต็มรูปแบบ
RISC-V Thread 1.3 Zigbee 3.0 Matter 5 µA Deep Sleep

AD8232
ECG / Heart Rate Monitor
Single-lead ECG AFE จาก Analog Devices วัดสัญญาณไฟฟ้าหัวใจ, มี instrumentation amplifier + HPF + LPF ในตัว ใช้ dry electrode ได้ ราคาถูก (~$10 บอร์ด breakout)
Interface: Analog · 3.3V · ~170 µA · SPI lead-off detect
MAX30102
Heart Rate & SpO2 Sensor (PPG)
Optical pulse oximeter จาก Maxim ใช้ LED แดง + อินฟราเรด วัด Heart Rate และ SpO2 ผ่าน I2C, มี ambient light cancellation ใช้ไฟน้อยมาก ~0.7 mA ระหว่างวัด เหมาะกับ wearable
I2C (0x57) · 1.8V–3.3V · 0.7mA · 18-bit ADC
SN65HVD230
CAN Bus Transceiver
CAN 2.0 transceiver จาก TI ที่จับคู่กับ ESP32 TWAI controller แปลง 3.3V logic จาก ESP32 เป็น differential CAN signals จำเป็นต้องมีเพราะ ESP32 TWAI ไม่สามารถต่อ CAN bus ตรงๆ ได้
3.3V · 1 Mbps · 110Ω termination · ISO 11898
MAX98357A
I2S Class-D Amplifier
Class-D mono amplifier ที่รับ I2S โดยตรงจาก ESP32 ให้กำลังขับ 3.2W ที่ 4Ω speaker, ไม่ต้องการ I2C/SPI config เหมาะมากสำหรับ smart speaker prototype ราคา ~$3 breakout
I2S · 2.5V–5.5V · 3.2W@4Ω · 92% efficiency
INMP441
MEMS I2S Microphone
Omnidirectional MEMS microphone ที่ output เป็น I2S SNR สูง 61 dBFS ใช้ร่วมกับ ESP32 ได้โดยตรง ใช้ใน voice assistant, audio recording, และ noise monitoring
I2S · 1.8V–3.3V · 61 dBFS SNR · –26 dBFS sensitivity
MPU-6050
6-Axis IMU (Accel + Gyro)
IMU ราคาถูกที่สุดที่ยังดีอยู่ 3-axis accelerometer + 3-axis gyroscope ใช้ใน wearable เพื่อ motion artifact removal บน ECG/PPG มี DMP (Digital Motion Processor) ใน chip ลด CPU load
I2C · 3.3V · 3.6mA · ±2g–±16g · ±250°/s–±2000°/s

C (ESP-IDF) modbus_rtu_master.c — ESP32 Modbus RTU Master 
// ESP32 Modbus RTU Master — อ่านค่าจาก RS-485 slave
// Hardware: ESP32 TX2→MAX485 DI, RX2→RO, GPIO4→DE/RE
// ESP-IDF: idf_component.yml → esp-modbus >=2.0.0

#include "esp_modbus_master.h"
#include "driver/uart.h"

#define UART_PORT    UART_NUM_2
#define UART_TX      17
#define UART_RX      16
#define RS485_EN     4       // DE/RE pin
#define SLAVE_ADDR   0x01
#define REG_START    0x0000  // Holding register เริ่มต้น
#define REG_COUNT    2

void modbus_read_task(void *arg) {
    // เริ่มต้น Modbus master (RTU mode)
    mb_communication_info_t comm_info = {
        .port     = UART_PORT,
        .mode     = MB_MODE_RTU,
        .baudrate = 9600,
        .parity   = UART_PARITY_DISABLE,
    };
    mbc_master_init(MB_PORT_SERIAL_MASTER, &comm_info);

    // กำหนด slave descriptor
    mb_slave_addr_entry_t slave = {
        .slave_addr = SLAVE_ADDR,
        .send_tout_ms = 150,
        .mb_param_offset = 0,
    };
    mbc_master_start();
    mbc_master_set_descriptor(&slave, 1);

    uint16_t reg_data[REG_COUNT];

    while (1) {
        // อ่าน Holding Registers 0x0000-0x0001 จาก slave
        esp_err_t err = mbc_master_send_request(
            MB_FUNC_READ_HOLDING_REGISTER,
            SLAVE_ADDR,
            REG_START,
            REG_COUNT,
            reg_data
        );

        if (err == ESP_OK) {
            printf("Slave %d → Reg[0]=%d, Reg[1]=%d\n",
                   SLAVE_ADDR, reg_data[0], reg_data[1]);
        } else {
            printf("Modbus error: %s\n", esp_err_to_name(err));
        }
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}
Arduino / ESP32 max30102_ble.ino — Heart Rate Monitor + BLE Notify 
// ESP32-C3 + MAX30102 Heart Rate Monitor → BLE Notify
// Library: SparkFun MAX3010x (Arduino), ESP32 BLE Arduino

#include <Wire.h>
#include "MAX30105.h"
#include "heartRate.h"
#include <BLEDevice.h>
#include <BLEServer.h>
#include <BLEUtils.h>
#include <BLE2902.h>

// BLE Heart Rate Service UUID (Bluetooth SIG Standard)
#define HR_SERVICE_UUID     "0000180D-0000-1000-8000-00805F9B34FB"
#define HR_CHAR_UUID        "00002A37-0000-1000-8000-00805F9B34FB"

MAX30105 particleSensor;
BLECharacteristic *hrChar;
const byte RATE_SIZE = 4;
byte rates[RATE_SIZE];
byte rateSpot = 0;
long lastBeat = 0;
float beatsPerMinute;
int beatAvg;

void setup() {
    Serial.begin(115200);
    Wire.begin(8, 9);  // SDA=GPIO8, SCL=GPIO9 (ESP32-C3)

    // เริ่ม MAX30102
    if (!particleSensor.begin(Wire, I2C_SPEED_FAST)) {
        Serial.println("MAX30102 not found!");
        while(1);
    }
    particleSensor.setup();
    particleSensor.setPulseAmplitudeRed(0x0A);

    // เริ่ม BLE
    BLEDevice::init("ESP32-HeartRate");
    BLEServer *server = BLEDevice::createServer();
    BLEService *hrService = server->createService(HR_SERVICE_UUID);
    hrChar = hrService->createCharacteristic(
        HR_CHAR_UUID,
        BLECharacteristic::PROPERTY_NOTIFY
    );
    hrChar->addDescriptor(new BLE2902());
    hrService->start();
    server->getAdvertising()->start();
}

void loop() {
    long irValue = particleSensor.getIR();

    if (checkForBeat(irValue)) {
        long delta = millis() - lastBeat;
        lastBeat = millis();
        beatsPerMinute = 60 / (delta / 1000.0);

        if (beatsPerMinute > 20 && beatsPerMinute < 255) {
            rates[rateSpot++ % RATE_SIZE] = (byte)beatsPerMinute;
            beatAvg = 0;
            for (byte x=0; x<RATE_SIZE; x++) beatAvg += rates[x];
            beatAvg /= RATE_SIZE;

            // ส่งผ่าน BLE (Heart Rate Measurement format)
            uint8_t hrData[2] = { 0x00, (uint8_t)beatAvg };
            hrChar->setValue(hrData, 2);
            hrChar->notify();

            Serial.printf("BPM=%.1f Avg=%d\n", beatsPerMinute, beatAvg);
        }
    }
}
