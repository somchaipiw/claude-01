Capacitive Soil Moisture + ESP32-C6 ส่ง Matter attribute
C++ (Arduino / ESP-IDF) ESP32-C6 + SEN0193 + Matter 1.5 
// Matter 1.5 Soil Moisture Sensor — ESP32-C6
#include <esp_matter.h>
#include <esp_matter_cluster.h>

#define SOIL_PIN         34    // Analog pin
#define DRY_VALUE        3200  // ADC ดิน dry
#define WET_VALUE        1200  // ADC ดิน wet

using namespace chip;
using namespace esp_matter;

node_t *node;
endpoint_t *soil_endpoint;
cluster_t *soil_cluster;

uint8_t readSoilPercent() {
  int raw = analogRead(SOIL_PIN);
  return constrain(
    map(raw, DRY_VALUE, WET_VALUE, 0, 100),
    0, 100
  );
}

void setup() {
  Serial.begin(115200);

  // สร้าง Matter node
  node::config_t node_config;
  node = node::create(&node_config, nullptr, nullptr);

  // สร้าง Soil Moisture endpoint (Matter 1.5)
  endpoint::soil_moisture_sensor::config_t soil_config;
  soil_config.soil_moisture_measurement.measured_value = 5000; // 50.00%
  soil_endpoint = endpoint::soil_moisture_sensor::create(
    node, &soil_config, ENDPOINT_FLAG_NONE, nullptr
  );

  esp_matter::start(nullptr);
}

void loop() {
  static unsigned long lastRead = 0;
  if (millis() - lastRead > 30000) { // ทุก 30 วินาที
    lastRead = millis();
    uint8_t pct = readSoilPercent();

    // อัปเดต Matter attribute (หน่วยเป็น 0.01%)
    attribute::update(
      endpoint::get_id(soil_endpoint),
      SoilMoistureMeasurement::Id,
      SoilMoistureMeasurement::Attributes::MeasuredValue::Id,
      pct * 100
    );
    Serial.printf("Soil: %d%%\n", pct);
  }
}
ECG Monitor — ESP32 + AD8232 + แสดงผล Serial Plotter
C++ (Arduino) ESP32 + AD8232 
#define ECG_PIN   34
#define LO_PLUS  32
#define LO_MINUS 33

void setup() {
  Serial.begin(115200);
  pinMode(LO_PLUS,  INPUT);
  pinMode(LO_MINUS, INPUT);
}

void loop() {
  if (digitalRead(LO_PLUS) ||
      digitalRead(LO_MINUS)) {
    Serial.println(0); // lead off
  } else {
    Serial.println(analogRead(ECG_PIN));
  }
  delay(10); // 100 Hz sample rate
}
Light Pollution Monitor — TSL2591 + ESP32-C3
C++ (Arduino) ESP32-C3 + TSL2591 
#include "Adafruit_TSL2591.h"

Adafruit_TSL2591 tsl(2591);

void setup() {
  Serial.begin(115200);
  tsl.begin();
  tsl.setGain(TSL2591_GAIN_HIGH);
  tsl.setTiming(TSL2591_INTEGRATIONTIME_300MS);
}

void loop() {
  sensors_event_t event;
  tsl.getEvent(&event);
  if (event.light) {
    Serial.printf("%.4f lux\n", event.light);
    // Bortle scale: <0.01 = Class 1 (dark)
    //              >100   = Class 9 (city)
  }
  delay(60000); // log ทุก 1 นาที
}
