Water Quality — Read pH + TDS + Turbidity C++ / Arduino 
// ESP32 Water Quality Sensor Node
// pH (analog), TDS (analog), Turbidity (analog)

#include <Arduino.h>
#include <WiFi.h>

const int PIN_PH         = 34;
const int PIN_TDS        = 35;
const int PIN_TURBIDITY  = 32;
const float VREF        = 3.3;
const int ADC_RES       = 4095;

// pH calibration: slope/offset จาก Buffer 4.0 & 7.0
float readPH(int pin) {
  float v = analogRead(pin) * VREF / ADC_RES;
  return 7.0 + ((v - 1.65) / -0.18); // slope ตามใบ calibration
}

// TDS ≈ Conductivity × 0.5 (NaCl approximation)
float readTDS(int pin, float tempC = 25.0) {
  float v = analogRead(pin) * VREF / ADC_RES;
  float compCoeff = 1.0 + 0.02 * (tempC - 25.0);
  float compV = v / compCoeff;
  return (133.42 * compV * compV * compV
         - 255.86 * compV * compV
         + 857.39 * compV) * 0.5;
}

// Turbidity: voltage สูง = ใส, ต่ำ = ขุ่น (NTU)
float readTurbidityNTU(int pin) {
  float v = analogRead(pin) * VREF / ADC_RES;
  if (v >= 2.5) return 0;
  return -1120.4 * v * v + 5742.3 * v - 4352.9;
}

void setup() {
  Serial.begin(115200);
  analogReadResolution(12);
}

void loop() {
  float ph  = readPH(PIN_PH);
  float tds = readTDS(PIN_TDS);
  float ntu = readTurbidityNTU(PIN_TURBIDITY);

  Serial.printf("pH=%.2f | TDS=%.1f ppm | Turbidity=%.1f NTU\n",
                ph, tds, ntu);
  delay(2000);
}
RS485 NPK Soil Sensor — Modbus RTU Read C++ / Arduino 
// ESP32 + MAX485 Module อ่าน NPK Sensor ผ่าน Modbus RTU
#include <HardwareSerial.h>

const int RS485_RX  = 16;
const int RS485_TX  = 17;
const int RS485_DE  = 4;  // DE/RE pin

HardwareSerial rs485(1);

// Modbus RTU command: อ่าน N,P,K,Temp,Humidity (5 registers)
byte cmd[] = {0x01, 0x03, 0x00, 0x00, 0x00, 0x05, 0x85, 0xC9};

void setup() {
  Serial.begin(115200);
  rs485.begin(9600, SERIAL_8N1, RS485_RX, RS485_TX);
  pinMode(RS485_DE, OUTPUT);
}

void loop() {
  // Send query
  digitalWrite(RS485_DE, HIGH);
  rs485.write(cmd, sizeof(cmd));
  rs485.flush();
  digitalWrite(RS485_DE, LOW);

  delay(200);  // รอ sensor ตอบกลับ

  byte buf[15] = {0};
  int n = rs485.readBytes(buf, 15);

  if (n >= 13) {
    int N    = (buf[3] << 8) | buf[4];
    int P    = (buf[5] << 8) | buf[6];
    int K    = (buf[7] << 8) | buf[8];
    float T  = ((buf[9]  << 8) | buf[10]) / 10.0;
    float H  = ((buf[11] << 8) | buf[12]) / 10.0;

    Serial.printf("N=%d P=%d K=%d mg/kg | T=%.1f°C H=%.1f%%\n",
                  N, P, K, T, H);
  }
  delay(3000);
}
MAX30102 BLE Heart Rate — XIAO ESP32-C6 C++ / Arduino 
// MAX30102 บน XIAO ESP32-C6 — BLE Advertising ส่ง Heart Rate
#include <Wire.h>
#include <MAX30105.h>
#include <heartRate.h>
#include <NimBLEDevice.h>

MAX30105 particleSensor;
NimBLEAdvertising *pAdv;

byte rates[4];
byte rateSpot = 0;
long lastBeat = 0;
float beatsPerMinute;
int beatAvg;

void setup() {
  Serial.begin(115200);
  particleSensor.begin(Wire, I2C_SPEED_FAST);
  particleSensor.setup();

  NimBLEDevice::init("EdgeCare-C6");
  pAdv = NimBLEDevice::getAdvertising();
}

void loop() {
  long irValue = particleSensor.getIR();
  if (checkForBeat(irValue)) {
    long delta = millis() - lastBeat;
    lastBeat = millis();
    beatsPerMinute = 60 / (delta / 1000.0);
    if (beatsPerMinute > 20 && beatsPerMinute < 255) {
      rates[rateSpot++ % 4] = beatsPerMinute;
      beatAvg = (rates[0]+rates[1]+rates[2]+rates[3]) / 4;
    }
  }
  // BLE Advertising ทุก 1 วินาที
  static long lastAdv = 0;
  if (millis() - lastAdv > 1000) {
    NimBLEAdvertisementData advData;
    advData.setName("EdgeCare");
    advData.setManufacturerData(String((char)beatAvg));
    pAdv->setAdvertisementData(advData);
    pAdv->start(1000); // advertise 1 วินาที
    lastAdv = millis();
  }
}
