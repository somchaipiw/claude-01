ESP32-S3 + SimpleFOC

Field-Oriented Control (FOC) บน ESP32 — ควบคุม BLDC Motor แบบ Professional

Espressif ออก BLDC Solution อย่างเป็นทางการรองรับ SimpleFOC Library บน ESP32-S3 และ ESP32-C3 แล้ว ระบบ FOC ให้การควบคุม Torque, Speed และ Position ที่แม่นยำโดยไม่ต้องใช้ DSP ชิปแยก รองรับ Speed Loop, Angle Loop และ Torque Loop ผ่าน Arduino IDE ด้วย SimpleFOC v3.x

  • ESP32-S3 ใช้ MCPWM (Motor Control PWM) hardware สำหรับ 3-Phase gate drive
  • จับคู่กับ TMC6300 (3-Phase gate driver สำหรับ BLDC ขนาดเล็ก) หรือ DRV8313
  • SparkFun IoT Brushless Motor Driver รวม ESP32-WROOM + TMC6300 ในบอร์ดเดียว
  • ใช้งานได้กับ Gimbal Motor, Drone Motor และ Robotic Joint ขนาด 2208–4114
🔗 Espressif BLDC Solution  ·  SparkFun IoT Brushless Driver

BLE Mesh

🌐 BLE Mesh Power Monitoring Network ด้วย ESP32-C6 + INA260

ตาข่าย Sensor Node ที่ใช้ ESP32-C6 แต่ละ Node วัดแรงดันและกระแสด้วย INA260 (I2C) และส่งข้อมูลผ่าน BLE Mesh ไปยัง Raspberry Pi Gateway รองรับ Deep Sleep 30 วินาที เหมาะกับ Smart Grid, Solar Monitoring และ Building Energy Management

  • Mesh: ESP-BLE-Mesh Stack บน ESP-IDF v5.x

  • Gateway: Raspberry Pi 5 รับข้อมูลและส่ง MQTT → InfluxDB

Code Guide — ESP32 + RC522 RFID Access Control พื้นฐาน

Arduino / C++

อ่าน RFID Card UID ด้วย RC522 + ESP32 (SPI)

ติดตั้ง Library: MFRC522 by GithubCommunity ผ่าน Arduino Library Manager · การต่อขา: SDA→GPIO5, SCK→GPIO18, MOSI→GPIO23, MISO→GPIO19, RST→GPIO27

esp32_rfid_access.ino Arduino C++ 
// ESP32 + RC522 RFID Access Control
// Library: MFRC522 by GithubCommunity

#include <SPI.h>
#include <MFRC522.h>

#define SS_PIN   5
#define RST_PIN  27
#define RELAY_PIN 4   // ต่อ Relay เปิด/ปิดประตู

MFRC522 rfid(SS_PIN, RST_PIN);

// รายการ UID ที่ได้รับอนุญาต (แก้ไขตาม Card จริง)
const byte authorizedCards[][4] = {
  {0xDE, 0xAD, 0xBE, 0xEF},  // Card 1
  {0xA1, 0xB2, 0xC3, 0xD4}   // Card 2
};
const int numCards = sizeof(authorizedCards) / 4;

void setup() {
  Serial.begin(115200);
  SPI.begin();
  rfid.PCD_Init();
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW);
  Serial.println("RFID Reader Ready — แตะบัตร");
}

bool isAuthorized(byte* uid, byte size) {
  if (size != 4) return false;
  for (int i = 0; i < numCards; i++) {
    if (memcmp(uid, authorizedCards[i], 4) == 0)
      return true;
  }
  return false;
}

void loop() {
  if (!rfid.PICC_IsNewCardPresent() || !rfid.PICC_ReadCardSerial())
    return;

  Serial.print("UID: ");
  for (byte i = 0; i < rfid.uid.size; i++) {
    Serial.printf("%02X ", rfid.uid.uidByte[i]);
  }
  Serial.println();

  if (isAuthorized(rfid.uid.uidByte, rfid.uid.size)) {
    Serial.println("✅ อนุญาต — เปิดประตู");
    digitalWrite(RELAY_PIN, HIGH);
    delay(3000);  // เปิด 3 วินาที
    digitalWrite(RELAY_PIN, LOW);
  } else {
    Serial.println("❌ ไม่อนุญาต — บัตรไม่ถูกต้อง");
  }

  rfid.PICC_HaltA();
  rfid.PCD_StopCrypto1();
}
หมายเหตุความปลอดภัย

ระบบ RFID แบบ UID-only ถือว่าความปลอดภัยต่ำ

สำหรับการใช้งานจริงที่ต้องการความปลอดภัยสูง ควรใช้ Encrypted MIFARE DESFire, Mutual Authentication หรือ Backend Verification แทนการตรวจสอบ UID อย่างเดียว — UID สามารถ Clone ได้ด้วย Hardware ราคาถูก
