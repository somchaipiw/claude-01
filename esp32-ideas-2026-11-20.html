เปรียบเทียบ ESP32-C6 vs ESP32-H2 สำหรับงาน Smart Home Protocol

คุณสมบัติ ESP32-C6 ESP32-H2 เหมาะกับ

Bluetooth LE BLE 5.0 BLE 5.3 H2 มี BLE ใหม่กว่า
IEEE 802.15.4 Thread + Zigbee Thread + Zigbee ทั้งคู่เหมือนกัน

ราคา (approx.) สูงกว่าเล็กน้อย ถูกกว่า H2 ถ้าไม่ต้องการ Wi-Fi

1. ESP32-C6 — ตั้งค่า Thread Network ด้วย ESP-IDF

ใช้ OpenThread stack ที่มาพร้อมกับ ESP-IDF v5.x สำหรับสร้าง Thread End Device

thread_init.c — ESP32-C6 Thread End Device 
// ESP-IDF v5.x — Thread End Device บน ESP32-C6
// idf_component.yml: requires esp_openthread, nvs_flash

#include "esp_openthread.h"
#include "esp_openthread_netif_glue.h"
#include "openthread/thread.h"
#include "openthread/dataset.h"
#include "nvs_flash.h"

// Thread Network dataset (ได้จาก Border Router)
static const char *DATASET_TLVS =
    "0E080000000000010000...";  // hex TLV จาก OTBR

void ot_task_worker(void *arg) {
    esp_openthread_platform_config_t config = {
        .radio_config = ESP_OPENTHREAD_DEFAULT_RADIO_CONFIG(),
        .host_config = ESP_OPENTHREAD_DEFAULT_HOST_CONFIG(),
        .port_config = ESP_OPENTHREAD_DEFAULT_PORT_CONFIG(),
    };

    // เริ่ม OpenThread stack
    esp_openthread_init(&config);

    // Apply network dataset
    otOperationalDatasetTlvs dataset;
    otDatasetParseTlvs(DATASET_TLVS, &dataset);
    otDatasetSetActiveTlvs(esp_openthread_get_instance(), &dataset);

    // เปิด Thread interface
    otIp6SetEnabled(esp_openthread_get_instance(), true);
    otThreadSetEnabled(esp_openthread_get_instance(), true);

    esp_openthread_launch_mainloop();
}

void app_main(void) {
    nvs_flash_init();
    // Thread ต้องการ 8KB stack
    xTaskCreate(ot_task_worker, "ot", 8192, NULL, 5, NULL);
}
หมายเหตุ: ESP32-C6 ใช้ radio เดียวกันสำหรับ Wi-Fi 6 และ IEEE 802.15.4 — ไม่สามารถใช้พร้อมกันได้พร้อมกันทันที ต้องใช้ Coexistence API หรือ time-sharing ที่ Espressif กำลัง develop อยู่

2. ESP32 Modbus RS485 — อ่านค่า sensor ผ่าน MAX485

ใช้ ESP-Modbus component (esp-idf) อ่านค่าอุณหภูมิจาก RS485 temperature transmitter

modbus_rs485_read.c — Modbus RTU Master 
// ESP32 Modbus RTU Master — อ่าน SHT20-485 temperature sensor
// Hardware: ESP32 + MAX485 module
// UART2: TX=GPIO17, RX=GPIO16, DE/RE=GPIO5

#include "mbc_master.h"
#include "esp_modbus_common.h"

#define MB_PORT_NUM     2
#define MB_DEV_SPEED    9600
#define MB_SLAVE_ADDR   1
#define MB_TEMP_REG     0x0001  // Input Register 1 = Temperature x10
#define MB_HUM_REG      0x0002  // Input Register 2 = Humidity x10

void modbus_read_task(void *arg) {
    mb_communication_info_t comm_info = {
        .port     = MB_PORT_NUM,
        .baudrate = MB_DEV_SPEED,
        .parity   = MB_PARITY_NONE,
        .mode     = MB_MODE_RTU,
    };

    mbc_master_init(MB_PORT_RS485_SERIAL, &comm_info);
    mbc_master_start();

    uint16_t regs[2];
    mb_param_request_t req = {
        .slave_addr = MB_SLAVE_ADDR,
        .command    = MB_FUNC_READ_INPUT_REGISTER,
        .reg_start  = MB_TEMP_REG,
        .reg_size   = 2
    };

    while(1) {
        esp_err_t err = mbc_master_send_request(&req, regs);
        if(err == ESP_OK) {
            float temp = regs[0] / 10.0f;
            float hum  = regs[1] / 10.0f;
            printf("Temp: %.1f°C  Hum: %.1f%%\n", temp, hum);
        }
        vTaskDelay(pdMS_TO_TICKS(5000));  // poll ทุก 5 วินาที
    }
}

3. ESP32-S3 — รับเสียงจาก INMP441 ผ่าน I2S

ตั้งค่า I2S driver สำหรับรับ 16-bit audio จาก MEMS microphone INMP441

i2s_mic_input.c — INMP441 on ESP32-S3 
// ESP32-S3 — INMP441 I2S MEMS Microphone
// Pins: SCK=GPIO41, WS=GPIO42, SD=GPIO2, L/R=GND (Left ch)

#include "driver/i2s_std.h"  // ESP-IDF v5+ new I2S API

#define I2S_SAMPLE_RATE  16000
#define I2S_BUF_LEN      1024

i2s_chan_handle_t rx_chan;

void i2s_mic_init(void) {
    i2s_chan_config_t chan_cfg = I2S_CHANNEL_DEFAULT_CONFIG(
        I2S_NUM_AUTO, I2S_ROLE_MASTER);
    i2s_new_channel(&chan_cfg, NULL, &rx_chan);

    i2s_std_config_t std_cfg = {
        .clk_cfg  = I2S_STD_CLK_DEFAULT_CONFIG(I2S_SAMPLE_RATE),
        .slot_cfg = I2S_STD_MSB_SLOT_DEFAULT_CONFIG(
            I2S_DATA_BIT_WIDTH_32BIT,  // INMP441 ส่ง 32bit frame
            I2S_SLOT_MODE_MONO),
        .gpio_cfg = {
            .mclk = I2S_GPIO_UNUSED,
            .bclk = 41,  // SCK
            .ws   = 42,  // WS (LRCLK)
            .dout = I2S_GPIO_UNUSED,
            .din  = 2,   // SD (data)
        },
    };
    i2s_channel_init_std_mode(rx_chan, &std_cfg);
    i2s_channel_enable(rx_chan);
}

void audio_record_task(void *arg) {
    int32_t raw[I2S_BUF_LEN];
    size_t bytes_read;

    while(1) {
        i2s_channel_read(rx_chan, raw,
            sizeof(raw), &bytes_read, 1000);

        // INMP441 data อยู่ใน bits 31-14 (shift right 14)
        for(int i=0; i < bytes_read/4; i++) {
            int16_t sample = (int16_t)(raw[i] >> 14);

        }
    }
}
เคล็ดลับ: INMP441 ส่งข้อมูลใน 24-bit ภายใน frame 32-bit โดย data จะอยู่ที่ bits[31:8] ต้อง shift right 8 บิต (ไม่ใช่ 14) หากต้องการ 24-bit precision ตรวจสอบ datasheet ของ INMP441 อีกครั้งก่อนใช้งานจริง
