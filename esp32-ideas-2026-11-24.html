1. ESP32-S3 — รัน TensorFlow Lite model (image inference)

ใช้ ESP-DL หรือ esp32-camera + TFLite Micro สำหรับ inference บน XIAO ESP32S3-Sense

C++ / Arduino 
// TFLite Micro image inference on ESP32-S3
#include "esp_camera.h"
#include "tensorflow/lite/micro/all_ops_resolver.h"
#include "tensorflow/lite/micro/micro_interpreter.h"
#include "model_data.h"  // model.tflite converted to C array

const int kArenaSize = 150 * 1024;  // 150 KB tensor arena
uint8_t tensor_arena[kArenaSize];

tflite::AllOpsResolver resolver;
const tflite::Model* model;
tflite::MicroInterpreter* interpreter;

void setup() {
  Serial.begin(115200);
  initCamera();  // init OV2640/OV5640

  model = tflite::GetModel(g_model_data);
  static tflite::MicroInterpreter static_interpreter(
    model, resolver, tensor_arena, kArenaSize);
  interpreter = &static_interpreter;
  interpreter->AllocateTensors();
}

void loop() {
  camera_fb_t* fb = esp_camera_fb_get();
  if (!fb) return;

  // Copy resized 64x64 grayscale into input tensor
  TfLiteTensor* input = interpreter->input(0);
  resizeAndCopyToTensor(fb->buf, input->data.uint8, 64, 64);
  esp_camera_fb_return(fb);

  interpreter->Invoke();

  TfLiteTensor* output = interpreter->output(0);
  int label = argmax(output->data.uint8, 3);
  Serial.printf("Prediction: %s\n", labels[label]);
}
2. ESP32 — อ่านค่า NPK Sensor ผ่าน RS485 Modbus RTU

ต้องการ MAX485 TTL-RS485 module ต่อที่ UART2 (GPIO16/17)

C++ / Arduino 
// ESP32 + MAX485 → RS485 NPK 7-in-1 Soil Sensor
#define DE_PIN    4   // MAX485 DE/RE control
#define RXD2     16
#define TXD2     17

HardwareSerial rs485(2);

// Modbus RTU request: อ่าน 7 register จาก address 0x01
const uint8_t query[] = {0x01, 0x03, 0x00, 0x00, 0x00, 0x07, 0x04, 0x08};

void setup() {
  Serial.begin(115200);
  rs485.begin(9600, SERIAL_8N1, RXD2, TXD2);
  pinMode(DE_PIN, OUTPUT);
}

void readNPK() {
  digitalWrite(DE_PIN, HIGH);  // TX mode
  rs485.write(query, sizeof(query));
  rs485.flush();
  digitalWrite(DE_PIN, LOW);   // RX mode

  delay(200);
  uint8_t resp[19];
  int len = rs485.readBytes(resp, 19);
  if (len < 19) { Serial.println("No response"); return; }

  float moisture =  (resp[3]  << 8 | resp[4])  / 10.0;
  float temp     =  (resp[5]  << 8 | resp[6])  / 10.0;
  int   nitrogen  =  (resp[7]  << 8 | resp[8]);
  int   phosphorus = (resp[9]  << 8 | resp[10]);
  int   potassium  = (resp[11] << 8 | resp[12]);
  float ph        = (resp[13] << 8 | resp[14]) / 10.0;

  Serial.printf("Moisture:%.1f%% Temp:%.1f°C N:%d P:%d K:%d pH:%.1f\n",
                moisture, temp, nitrogen, phosphorus, potassium, ph);
}

void loop() { readNPK(); delay(5000); }
3. ESP32 + LoRa SX1276 — ส่งข้อมูล Sensor ระยะไกล

ใช้ library LoRa by sandeep mistry ส่งค่า sensor ทุก 30 วินาที SX1276 ต่อผ่าน SPI (GPIO5/18/19/23)

C++ / Arduino 
#include <SPI.h>
#include <LoRa.h>

#define LORA_SS   5
#define LORA_RST  14
#define LORA_DIO0 2
#define FREQ      433E6   // 433 MHz สำหรับไทย

void setup() {
  Serial.begin(115200);
  LoRa.setPins(LORA_SS, LORA_RST, LORA_DIO0);
  if (!LoRa.begin(FREQ)) {
    Serial.println("LoRa init failed!");
    while (1);
  }
  LoRa.setSpreadingFactor(9);       // SF9 — balance range/speed
  LoRa.setSignalBandwidth(125E3);  // 125 kHz BW
  LoRa.setCodingRate4(5);          // CR 4/5
  Serial.println("LoRa ready");
}

void loop() {
  float moisture  = readMoisture();
  int   nitrogen  = readNitrogen();
  float temp      = readTemp();

  LoRa.beginPacket();
  LoRa.printf("{\"id\":1,\"m\":%.1f,\"n\":%d,\"t\":%.1f}",
              moisture, nitrogen, temp);
  LoRa.endPacket();

  Serial.printf("Sent: m=%.1f n=%d t=%.1f\n", moisture, nitrogen, temp);
  esp_deep_sleep(30e6);  // deep sleep 30 วินาที ประหยัดแบต
}
