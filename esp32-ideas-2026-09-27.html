DS18B20 Multi-Sensor บน 1-Wire — อ่านค่า 10 ตัวพร้อมกัน

ใช้ library OneWire + DallasTemperature สแกน ROM address ทุก sensor และอ่านค่าแบบ async

C++ (Arduino / ESP32) DS18B20 Multi-Point 
#include <OneWire.h>
#include <DallasTemperature.h>

#define ONE_WIRE_PIN 4
OneWire ow(ONE_WIRE_PIN);
DallasTemperature sensors(&ow);

void setup() {
  Serial.begin(115200);
  sensors.begin();
  sensors.setResolution(12);          // 12-bit = 0.0625°C
  sensors.setWaitForConversion(false); // async mode
  Serial.printf("Found %d sensors\n", sensors.getDeviceCount());
}

void loop() {
  sensors.requestTemperatures(); // kick conversion
  delay(750);                      // wait 12-bit conversion

  int count = sensors.getDeviceCount();
  for (int i = 0; i < count; i++) {
    float temp = sensors.getTempCByIndex(i);
    Serial.printf("Sensor[%d]: %.2f°C\n", i, temp);
  }
  delay(30000); // อ่านทุก 30 วิ
}

SCD41 อ่านค่า CO₂ ด้วย I2C — พร้อม Low-Power Periodic Measurement

ใช้ Sensirion Arduino library Sensirion I2C SCD4x อ่าน CO₂/Temp/RH ทุก 5 วินาที พร้อม low-power mode สำหรับ battery device

C++ (Arduino / ESP32) SCD41 CO₂ Sensor 
#include <Wire.h>
#include <SensirionI2CScd4x.h>

SensirionI2CScd4x scd4x;

void setup() {
  Serial.begin(115200);
  Wire.begin();
  scd4x.begin(Wire);
  scd4x.stopPeriodicMeasurement();  // reset if was running
  scd4x.startLowPowerPeriodicMeasurement(); // อ่านทุก 30 วิ กินไฟน้อย
}

void loop() {
  bool isReady = false;
  scd4x.getDataReadyFlag(isReady);

  if (isReady) {
    uint16_t co2;
    float temp, humidity;
    scd4x.readMeasurement(co2, temp, humidity);
    Serial.printf("CO2: %u ppm | Temp: %.1f°C | RH: %.1f%%\n",
                  co2, temp, humidity);
  }
  delay(5000);
}

Hydroponics pH Control — อ่าน Atlas EZO-pH + ควบคุม Peristaltic Pump

ตรวจ pH น้ำ Hydroponic ทุก 15 นาที ถ้าออกนอก range 5.8–6.2 จะสั่ง pump เติม pH up/down

C++ (Arduino / ESP32) pH Control Loop 
#include <Wire.h>
#define EZO_PH_ADDR  0x63  // default I2C address
#define PUMP_UP_PIN  18    // pH UP pump relay
#define PUMP_DN_PIN  19    // pH DOWN pump relay

float readPH() {
  Wire.beginTransmission(EZO_PH_ADDR);
  Wire.write("R");             // send Read command
  Wire.endTransmission();
  delay(900);               // EZO needs 900ms
  Wire.requestFrom(EZO_PH_ADDR, 20);
  char buf[20] = {0};
  int i = 0;
  while (Wire.available()) buf[i++] = Wire.read();
  return atof(buf + 1); // skip status byte
}

void loop() {
  float ph = readPH();
  Serial.printf("pH: %.3f\n", ph);

  if (ph < 5.8) {
    digitalWrite(PUMP_UP_PIN, HIGH); delay(500);
    digitalWrite(PUMP_UP_PIN, LOW);
    Serial.println("pH low — added pH UP");
  } else if (ph > 6.2) {
    digitalWrite(PUMP_DN_PIN, HIGH); delay(500);
    digitalWrite(PUMP_DN_PIN, LOW);
    Serial.println("pH high — added pH DOWN");
  }
  delay(900000); // ตรวจทุก 15 นาที
}
