ESP32-E22: ชิป Wi-Fi 6E ตัวแรกจาก Espressif

Espressif เปิดตัว ESP32-E22 ที่ CES 2026 ในฐานะ connectivity co-processor ตัวแรกของตระกูลที่รองรับ Wi-Fi 6E (Tri-Band: 2.4 / 5 / 6 GHz) พร้อม 2×2 MIMO และแบนด์วิดธ์ 160 MHz ช่วยให้ส่งข้อมูลได้เร็วถึง 2.1 Gbps ในการทดสอบ iperf

ตัวอย่าง Code ESP32-C6 Smart Energy Monitor + Matter
Arduino / C++
// ESP32-C6 Smart Energy Monitor with Matter 1.5
// Sensors: HLW8012 (current/power) + CT Sensor (SCT-013)
// Reports energy usage via Matter ElectricalMeasurement cluster

#include <Arduino.h>
#include <Matter.h>
#include <HLW8012.h>

#define SEL_PIN   5
#define CF1_PIN   6
#define CF_PIN    7
#define RELAY_PIN 8

HLW8012 hlw;
MatterEnergyMonitor energyMonitor;

float voltage_V, current_A, power_W, energy_kWh;
const float OVERCURRENT_LIMIT = 16.0; // amps

void IRAM_ATTR hlwCF1Interrupt()  { hlw.cf1_interrupt(); }
void IRAM_ATTR hlwCFInterrupt()   { hlw.cf_interrupt(); }

void setup() {
  Serial.begin(115200);
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, HIGH); // Circuit ON

  // Init HLW8012 energy measurement chip
  hlw.begin(CF1_PIN, CF_PIN, SEL_PIN, HIGH, true);
  hlw.setResistors(0.001, 470000, 1000);
  attachInterrupt(digitalPinToInterrupt(CF1_PIN), hlwCF1Interrupt, FALLING);
  attachInterrupt(digitalPinToInterrupt(CF_PIN),  hlwCFInterrupt,  FALLING);

  // Init Matter over Thread/Wi-Fi on ESP32-C6
  Matter.begin();
  energyMonitor.begin("Smart Breaker", "ESP32-C6");
  Serial.println("Matter pairing code: " + Matter.getManualPairingCode());
}

void loop() {
  voltage_V = hlw.getVoltage();
  current_A = hlw.getCurrent();
  power_W   = hlw.getActivePower();
  energy_kWh += power_W / 3600000.0;

  // Overcurrent protection: trip the breaker
  if (current_A > OVERCURRENT_LIMIT) {
    digitalWrite(RELAY_PIN, LOW);
    energyMonitor.reportFault("OVERCURRENT");
    Serial.printf("TRIPPED! %.2fA exceeded limit\n", current_A);
  }

  // Update Matter cluster every 10s
  static uint32_t lastReport = 0;
  if (millis() - lastReport > 10000) {
    energyMonitor.setVoltage(voltage_V);
    energyMonitor.setCurrent(current_A);
    energyMonitor.setPower(power_W);
    energyMonitor.setEnergyTotal(energy_kWh);
    lastReport = millis();
    Serial.printf("%.1fV  %.2fA  %.1fW  %.4f kWh\n",
      voltage_V, current_A, power_W, energy_kWh);
  }

  Matter.loop();
  delay(100);
}
