โฟกัสพิเศษ: ESP32 + UWB Indoor Positioning

ระบบระบุตำแหน่งในร่มความแม่นยำระดับเซนติเมตร ด้วย DWM3000/DW3000

UWB คืออะไร และทำไมต้องใช้กับ ESP32?

Ultra-Wideband (UWB) ใช้ Time of Flight (ToF) วัดระยะจากคลื่นวิทยุ ทำให้ได้ความแม่นยำ ±10 cm ซึ่งดีกว่า WiFi (±3 m) และ BLE (±1-3 m) มาก

โมดูล Qorvo DWM3000/DW3000 บน Makerfabs board ต่อกับ ESP32 ผ่าน SPI ทำให้สร้าง Real-Time Location System (RTLS) ได้ในราคาไม่แพง

UWB DWM3000 RTLS 10cm Accuracy
พารามิเตอร์ค่า
ความแม่นยำ~10 cm (ในร่ม)
ระยะทำงาน50-100 m (ในร่ม)
FrequencyChannel 5: 6.5 GHz / Ch9: 8 GHz
InterfaceSPI (ต่อกับ ESP32)
Anchors ขั้นต่ำ3 จุด (2D) / 4 จุด (3D)
AlgorithmDS-TWR (Double-Sided TWR)
Update rate~10 Hz
ราคา~$15/โมดูล (Makerfabs)
🏗️ สถาปัตยกรรมระบบ RTLS
Anchor (3-4 ตัว, ตำแหน่งคงที่):
ESP32 + DWM3000, flash firmware "anchor"
วางมุมห้อง ระยะรู้จุดพิกัดชัดเจน

Tag (1+ ตัว, ติดบนวัตถุ/คน):
ESP32 + DWM3000, firmware "tag"
ส่งสัญญาณ TWR ไปหา anchor ทั้งหมด

Visualization Server:
Python รับข้อมูลระยะ → คำนวณ Trilateration → plot 2D map แบบ real-time

ไอเดียโปรเจกต์ที่น่าสร้าง

โปรเจกต์ใหม่ที่ไม่ซ้ำกับที่เคยทำ — เน้น Smart Monitoring และ Industrial IoT

💧 ระบบตรวจคุณภาพน้ำ 6 พารามิเตอร์
สร้าง water quality monitor แบบ all-in-one:
• pH (Atlas EZO-pH หรือ SEN0161)
• Turbidity (SEN0189 ของ DFRobot)
• TDS / Conductivity (SEN0244)
• Dissolved Oxygen (EZO-DO, ±0.05 mg/L)
• Temperature (DS18B20 — ชดเชย DO reading)
• Water Level (ultrasonic)

⚙️ Predictive Maintenance สำหรับมอเตอร์ไฟฟ้า

💧 Water Quality: TDS + Turbidity + Temp
อ่านค่า sensor 3 ตัวพร้อมกัน และส่งไป Serial:
// Water Quality Monitor — ESP32 #include <OneWire.h> #include <DallasTemperature.h> #define TDS_PIN 34 #define TURB_PIN 35 #define DS18_PIN 4 #define VREF 3.3f OneWire oneWire(DS18_PIN); DallasTemperature ds18(&oneWire); float readTDS(float tempC) { float v = analogRead(TDS_PIN) / 4095.0 * VREF; float comp = 1.0 + 0.02 * (tempC - 25.0); return (v / comp) * 500; // ppm } float readTurbNTU() { float v = analogRead(TURB_PIN) / 4095.0 * VREF; // SEN0189: NTU = -1120.4*v^2 + 5742.3*v - 4352.9 return -1120.4*v*v + 5742.3*v - 4352.9; } void loop() { ds18.requestTemperatures(); float t = ds18.getTempCByIndex(0); float tds = readTDS(t); float ntu = readTurbNTU(); Serial.printf("T=%.1fC TDS=%.0fppm NTU=%.1f\n", t, tds, ntu); delay(2000); }
📊 Vibration FFT ด้วย MPU6050
สุ่มตัวอย่าง + FFT หา dominant frequency:
#include <Wire.h> #include <MPU6050.h> #include <arduinoFFT.h> const uint16_t N = 512; const float Fs = 1000.0; // Hz double vReal[N], vImag[N]; MPU6050 mpu; ArduinoFFT<double> FFT(vReal, vImag, N, Fs); void collectSamples() { for (int i = 0; i < N; i++) { int16_t ax, ay, az, gx, gy, gz; mpu.getMotion6(&ax, &ay, &az, &gx, &gy, &gz); vReal[i] = az / 16384.0; // g vImag[i] = 0; delayMicroseconds(1000); } } void loop() { collectSamples(); FFT.windowing(FFTWindow::Hann, FFTDirection::Forward); FFT.compute(FFTDirection::Forward); FFT.complexToMagnitude(); float peakHz = FFT.majorPeak(); Serial.printf("Peak: %.1f Hz\n", peakHz); // เตือนถ้า peak ตรงกับ bearing fault freq delay(500); }
🌱 อ่าน NPK Sensor ผ่าน RS485/Modbus
ใช้ MAX485 ต่อกับ ESP32 UART2:
// NPK Sensor — Modbus RTU via MAX485 #define DE_PIN 4 // MAX485 DE/RE // Query frame สำหรับ NPK sensor (N, P, K) byte query[] = { 0x01, 0x03, 0x00, 0x1E, 0x00, 0x03, 0x65, 0xCD }; void readNPK() { digitalWrite(DE_PIN, HIGH); // TX mode Serial2.write(query, sizeof(query)); Serial2.flush(); digitalWrite(DE_PIN, LOW); // RX mode delay(200); if (Serial2.available() >= 11) { byte res[11]; Serial2.readBytes(res, 11); int N_val = (res[3] << 8) | res[4]; int P_val = (res[5] << 8) | res[6]; int K_val = (res[7] << 8) | res[8]; Serial.printf("N=%d P=%d K=%d mg/kg\n", N_val, P_val, K_val); } }
