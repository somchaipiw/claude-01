ESP32 + LoRaWAN — Smart Agriculture ระยะไกล ไม่ต้องพึ่ง 4G

LoRaWAN ส่งข้อมูลได้ไกลกว่า 10 กม. ด้วยแบต AA สองก้อน ทำงานได้หลายปี — เหมาะกับฟาร์มขนาดใหญ่ที่ไม่มีสัญญาณมือถือ ESP32 ทำหน้าที่เป็น Gateway เชื่อม Sensor Node → LoRa → Cloud เปิดทางให้เกษตรกรไทยมีระบบ IoT ราคาไม่แพง

// ESP32 Solar Sensor Node — Deep Sleep + LoRa (Arduino Framework)
// Hardware: TTGO LoRa32 V2, Capacitive Soil Moisture Sensor, DS18B20

#include <Arduino.h>
#include <LoRa.h>
#include <OneWire.h>
#include <DallasTemperature.h>

// LoRa pins for TTGO LoRa32
#define LORA_SCK   5
#define LORA_MISO  19
#define LORA_MOSI  27
#define LORA_SS    18
#define LORA_RST   14
#define LORA_DI0   26

#define SOIL_PIN      34  // ADC pin สำหรับ Capacitive Soil Moisture
#define TEMP_PIN      13  // DS18B20 Data pin
#define SLEEP_SEC     (15 * 60)  // 15 นาที

OneWire oneWire(TEMP_PIN);
DallasTemperature tempSensor(&oneWire);

float readSoilMoisture() {
  int raw = analogRead(SOIL_PIN);
  // 4095 = แห้งสนิท, 1500 = เปียกมาก (ปรับตาม Sensor จริง)
  return map(raw, 4095, 1500, 0, 100);
}

void setup() {
  Serial.begin(115200);

  // เริ่มอ่าน Sensors
  tempSensor.begin();
  tempSensor.requestTemperatures();
  float tempC = tempSensor.getTempCByIndex(0);
  float soilPct = readSoilMoisture();

  // เริ่ม LoRa
  SPI.begin(LORA_SCK, LORA_MISO, LORA_MOSI, LORA_SS);
  LoRa.setPins(LORA_SS, LORA_RST, LORA_DI0);

  if (!LoRa.begin(915E6)) {  // 915 MHz (Asia)
    Serial.println("LoRa init failed");
    ESP.deepSleep(SLEEP_SEC * 1000000ULL);
    return;
  }

  // สร้าง Payload เป็น JSON
  String payload = "{";
  payload += "\"soil\":" + String(soilPct, 1);
  payload += ",\"temp\":" + String(tempC, 2);
  payload += "}";

  // ส่งผ่าน LoRa
  LoRa.beginPacket();
  LoRa.print(payload);
  LoRa.endPacket();

  Serial.printf("Sent: %s\n", payload.c_str());

  // ปิด LoRa แล้วเข้า Deep Sleep ประหยัดไฟ
  LoRa.end();
  delay(100);
  ESP.deepSleep(SLEEP_SEC * 1000000ULL);
}

void loop() {
  // ไม่ถึง loop เพราะ Deep Sleep รีสตาร์ท setup() ใหม่ทุกครั้ง
}
📦 Library ที่ต้องติดตั้ง: LoRa by Sandeep Mistry DallasTemperature OneWire — ติดตั้งผ่าน Arduino Library Manager
