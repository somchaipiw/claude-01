DIY Thermal Heatmap Camera — ESP32-WROVER + MLX90640 + Web Dashboard
สร้างกล้อง Thermal Imaging แบบ DIY ด้วย MLX90640 (32×24 pixels) เชื่อมต่อผ่าน I²C แล้ว stream frame ผ่าน WebSocket ไปแสดงผลเป็น heatmap บนเว็บบราวเซอร์แบบ real-time
  • Hardware: ESP32-WROVER + MLX90640 breakout + จ่ายไฟ 3.3V
  • Firmware: ใช้ library melexis/mlx90640-library, อ่านค่า 32×24=768 pixels ทุก ~100ms
  • Web UI: Canvas API + bilinear interpolation แสดง false-color (blue→red)
  • Use case: ตรวจหาจุดร้อนใน PCB, ตรวจผนังฉนวนรั่ว, นับคนใน room
📐 เปรียบเทียบเซ็นเซอร์: AMG8833 (8×8, ~$10) สำหรับฝึกหัด | MLX90640 (32×24, ~$18) สำหรับโปรเจกต์จริง | MLX90641 (16×12, ~$12) สมดุลราคา/คุณภาพ

💡 ระดับ Beginner · 1 วัน
mmWave Smart Lighting — ESP32 + HLK-LD2410 ไฟอัตโนมัติแม่นยำ 100%
สร้างระบบไฟอัตโนมัติที่ ไม่ดับทิ้งคนที่นั่งอยู่เฉยๆ (ปัญหาหลักของ PIR) ด้วย mmWave radar ตรวจจับ still presence ผ่าน UART parse และ relay ควบคุมหลอดไฟ
  • Wiring: LD2410B TX→GPIO16, RX→GPIO17 (UART2), OUT pin→GPIO4
  • Mode: ใช้ OUT pin (simple HIGH/LOW) สำหรับ basic presence, หรือ UART parse สำหรับ zone detection
  • Logic: debounce 5s ก่อนดับไฟ, ปรับ sensitivity ผ่าน BLE config app
  • Upgrade: ต่อ ESPHome integration เข้า Home Assistant ได้ทันที
🏠 Use cases: ห้องน้ำ (ไฟดับทิ้งคนที่ยืนนิ่งๆ), ห้องประชุม (auto-off), ห้องนอน (ไฟกลางคืนอัตโนมัติ)

☀️ ระดับ Advanced · 1 สัปดาห์
Solar-Harvested Sensor Node — ESP32 + BQ25570 + Deep Sleep ไม่ต้องชาร์จ
สร้าง outdoor sensor node ที่ทำงานจากพลังงานแสงอาทิตย์ล้วนๆ ใช้ BQ25570 PMIC (Ultra-Low Power Harvester) ชาร์จ supercapacitor แล้วปลุก ESP32 จาก deep sleep ทุก 15 นาที วัดข้อมูลและส่ง LoRa ก่อนกลับไปนอน
  • Panel: Solar cell เล็ก 0.5W, V_OC = 5–6V → BQ25570 input
  • Storage: Supercapacitor 10F หรือ LiPo 500mAh
  • MCU: ESP32 deep sleep mode (ใช้ไฟ ~10µA) ปลุกทุก 15 นาที
  • Sensor: BME280 (T/H/P) อ่านค่าใน 1 วินาที แล้ว shutdown
  • Radio: LoRa SX1276 ส่งข้อมูล 50 bytes ไป gateway
  • Power budget: 15 นาที/cycle = ใช้พลังงานเฉลี่ย <1mA
🔋 Energy Budget tip: Active time ~2 วินาที @ 150mA = 300mAs, Deep sleep 897 วินาที @ 0.01mA = 9mAs → รวม ~309mAs ต่อ cycle หรือเฉลี่ย 0.34mA — แผงโซลาร์ 0.5W ชาร์จได้สบาย

Code Guides — ตัวอย่างโค้ดที่เอาไปใช้ได้จริง

ตัวอย่างที่ 1: ESP32 + MLX90640 — อ่านค่า Thermal Array ผ่าน I²C และ stream WebSocket
// ESP32 + MLX90640 Thermal Camera (ต้องการ: Melexis MLX90640 library) // pip: https://github.com/melexis/mlx90640-library #include <Wire.h> #include <MLX90640_API.h> #include <WebSocketsServer.h> #include <ArduinoJson.h> #define MLX_ADDR 0x33 #define FRAME_RATE 4 // Hz (0.5, 1, 2, 4, 8, 16, 32, 64) float mlxFrame[768]; // 32x24 = 768 pixels float emissivity = 0.95; // ค่า emissivity ของวัสดุ (ผิวคน ~0.98) paramsMLX90640 mlxParams; WebSocketsServer ws(81); void setup() { Wire.begin(21, 22); Wire.setClock(400000); // Fast mode I2C 400kHz // โหลด calibration data จาก EEPROM ใน sensor uint16_t eeData[832]; MLX90640_DumpEE(MLX_ADDR, eeData); MLX90640_ExtractParameters(eeData, &mlxParams); // ตั้ง frame rate MLX90640_SetRefreshRate(MLX_ADDR, 0x03); // 0x03 = 4Hz ws.begin(); WiFi.begin("SSID", "PASSWORD"); } void loop() { ws.loop(); uint16_t rawData[834]; MLX90640_GetFrameData(MLX_ADDR, rawData); MLX90640_CalculateTo(rawData, &mlxParams, emissivity, 25.0, mlxFrame); // ส่ง 768 ค่าผ่าน WebSocket เป็น binary array ws.broadcastBIN((uint8_t*)mlxFrame, sizeof(mlxFrame)); delay(250); } // หมายเหตุ: ใช้ Canvas API ฝั่ง browser ทำ bilinear interpolation // และ map temperature → color (blue 20°C → red 40°C)
ตัวอย่างที่ 2: ESP32 + HLK-LD2410B — อ่าน UART และควบคุม Relay ตาม Presence Zone
// ESP32 + HLK-LD2410B mmWave Radar — UART Protocol Parser // Wiring: LD2410 TX → GPIO16, LD2410 RX → GPIO17 #define RADAR_RX 16 #define RADAR_TX 17 #define RELAY_PIN 26 HardwareSerial radarSerial(2); // UART2 struct RadarData { bool hasMotion; bool hasStill; uint8_t motionDist; // cm units (gate) uint8_t stillDist; uint8_t motionEnergy; uint8_t stillEnergy; }; RadarData parseLD2410Frame(uint8_t* buf, int len) { RadarData d = {false, false, 0, 0, 0, 0}; // LD2410 Basic Frame: FD FC FB FA [len_lo len_hi] [data] 04 03 02 01 if (len < 13) return d; if (buf[0] != 0xF4 || buf[1] != 0xF3) return d; d.hasMotion = (buf[8] == 0x01 || buf[8] == 0x03); d.hasStill = (buf[8] == 0x02 || buf[8] == 0x03); d.motionDist = buf[9]; // distance gate (1 gate ≈ 0.75m) d.motionEnergy = buf[10]; d.stillDist = buf[11]; d.stillEnergy = buf[12]; return d; } void setup() { radarSerial.begin(256000, SERIAL_8N1, RADAR_RX, RADAR_TX); pinMode(RELAY_PIN, OUTPUT); } unsigned long lastPresence = 0; const unsigned long OFF_DELAY = 10000; // ดับไฟหลัง 10 วินาที void loop() { if (radarSerial.available() > 12) { uint8_t buf[64]; int n = radarSerial.readBytes(buf, 64); RadarData r = parseLD2410Frame(buf, n); if (r.hasMotion || r.hasStill) { lastPresence = millis(); digitalWrite(RELAY_PIN, HIGH); // เปิดไฟ } } if (millis() - lastPresence > OFF_DELAY) { digitalWrite(RELAY_PIN, LOW); // ดับไฟ } }
