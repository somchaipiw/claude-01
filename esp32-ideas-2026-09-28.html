Protocol

ESP32-C5 รองรับ CAN FD — TWAI ใหม่ใน ESP-IDF v6.0

ESP32-C5 เป็นชิปแรกในตระกูล ESP32 ที่ TWAI controller รองรับ CAN FD (Flexible Data-rate) ตามมาตรฐาน ISO 11898-1 ทำงานได้ทั้ง classic CAN และ CAN FD frames ในขณะที่ ESP32/S3 รองรับแค่ Classic CAN ทำให้ ESP32-C5 เหมาะสำหรับงาน automotive/industrial ระดับสูงที่ต้องการ data rate สูงถึง 8 Mbit/s และ payload ใหญ่ขึ้น (64 bytes vs 8 bytes)

CAN FD ESP32-C5 8 Mbit/s 64B frame ESP-IDF v6

ESP32 TWAI + SN65HVD230 — อ่าน OBD-II PID (RPM, Speed) C++ / Arduino 
// CAN transceiver: SN65HVD230
// TX → GPIO21 | RX → GPIO22 | VCC → 3.3V
#include "driver/twai.h"

#define CAN_TX 21
#define CAN_RX 22
#define OBD_REQUEST_ID  0x7DF
#define OBD_RESPONSE_ID 0x7E8

void setup() {
  Serial.begin(115200);
  twai_general_config_t g_config =
    TWAI_GENERAL_CONFIG_DEFAULT((gpio_num_t)CAN_TX,
                                   (gpio_num_t)CAN_RX,
                                   TWAI_MODE_NORMAL);
  twai_timing_config_t  t_config = TWAI_TIMING_CONFIG_500KBITS();
  twai_filter_config_t  f_config = TWAI_FILTER_CONFIG_ACCEPT_ALL();
  twai_driver_install(&g_config, &t_config, &f_config);
  twai_start();
}

// ส่ง OBD-II request PID 0x0C (RPM)
void requestOBD(uint8_t pid) {
  twai_message_t msg = {};
  msg.identifier   = OBD_REQUEST_ID;
  msg.data_length_code = 8;
  msg.data[0] = 0x02; // 2 bytes follow
  msg.data[1] = 0x01; // Mode 01 (current data)
  msg.data[2] = pid;
  twai_transmit(&msg, pdMS_TO_TICKS(100));
}

void loop() {
  requestOBD(0x0C); // PID: RPM
  twai_message_t rx;
  if (twai_receive(&rx, pdMS_TO_TICKS(200)) == ESP_OK) {
    if (rx.identifier == OBD_RESPONSE_ID && rx.data[2] == 0x0C) {
      uint16_t rpm = ((rx.data[3] << 8) | rx.data[4]) / 4;
      Serial.printf("RPM: %d\n", rpm);
    }
  }
  delay(500);
}
HX711 Beehive Scale — อ่านน้ำหนักรังผึ้ง + ส่ง MQTT C++ / Arduino 
// Library: bogde/HX711-Arduino-Library
#include "HX711.h"
#include <WiFi.h>
#include <PubSubClient.h>

#define DOUT 4
#define CLK  5
const float CALIBRATION_FACTOR = -7050.0; // ปรับตาม load cell

HX711 scale;
WiFiClient espClient;
PubSubClient mqtt(espClient);

void setup() {
  scale.begin(DOUT, CLK);
  scale.set_scale(CALIBRATION_FACTOR);
  scale.tare(); // ศูนย์น้ำหนักเปล่า
  // WiFi.begin(...) + mqtt.connect(...)
}

void loop() {
  if (scale.is_ready()) {
    float weight = scale.get_units(10); // เฉลี่ย 10 ครั้ง (กิโลกรัม)
    char buf[16];
    snprintf(buf, sizeof(buf), "%.3f", weight);
    mqtt.publish("beehive/weight", buf);
    Serial.printf("น้ำหนักรัง: %.3f kg\n", weight);
  }
  // deep sleep ระหว่าง interval 5 นาที
  esp_sleep_enable_timer_wakeup(5 * 60 * 1000000ULL);
  esp_deep_sleep_start();
}
Atlas EZO-ORP I2C — อ่านค่า ORP สระว่ายน้ำ + ตัดสินใจ Chlorine Dosing C++ / Arduino 
// Atlas EZO-ORP default I2C address: 0x62
#include <Wire.h>

const uint8_t EZO_ORP_ADDR = 0x62;
const int    RELAY_CHLORINE = 26; // relay ต่อ Stenner pump
const float  ORP_TARGET     = 700.0; // mV เป้าหมาย

float readORP() {
  Wire.beginTransmission(EZO_ORP_ADDR);
  Wire.write("R");              // คำสั่งอ่าน
  Wire.endTransmission();
  delay(1000);                  // รอ conversion
  Wire.requestFrom(EZO_ORP_ADDR, (uint8_t)20);
  char buf[20] = {};
  uint8_t i = 0;
  while (Wire.available() && i < 19) buf[i++] = Wire.read();
  return atof(buf + 1); // skip status byte
}

void loop() {
  float orp = readORP();
  Serial.printf("ORP: %.1f mV\n", orp);

  if (orp < ORP_TARGET - 30) {
    // ORP ต่ำเกินไป → เปิด chlorine pump 10 วินาที
    digitalWrite(RELAY_CHLORINE, HIGH);
    delay(10000);
    digitalWrite(RELAY_CHLORINE, LOW);
  }
  delay(60000); // วัดทุก 1 นาที
}
